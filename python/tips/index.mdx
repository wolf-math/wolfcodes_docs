export default function TipsAccordion() {
  const [openIndexes, setOpenIndexes] = useState([0]);

  return (
    <ul className={styles.topicList}>
      {sections.map((section, index) => {
        const isOpen = openIndexes.includes(index);

        return (
          <li className={styles.topicItem} key={section.title}>
            <button
              type="button"
              className={styles.topicButton}
              aria-expanded={isOpen}
              aria-controls={`topic-panel-${index}`}
              onClick={() =>
                setOpenIndexes((current) =>
                  current.includes(index)
                    ? current.filter((item) => item !== index)
                    : [...current, index]
                )
              }
            >
              <span
                className={`${styles.chevron} ${isOpen ? styles.chevronOpen : ''}`}
                aria-hidden="true"
              >
                ▸
              </span>
              
              <span className={styles.topicTitle}>{section.title}</span>
              <span className={styles.topicMeta}>{section.lessons.length} lessons</span>
            </button>

            {isOpen ? (
              <div className={styles.topicContent} id={`topic-panel-${index}`}>
                <p className={styles.topicDescription}>{section.description}</p>

                <ul className={styles.lessonList}>
                  {section.lessons.map(([label, to]) => (
                    <li key={to}>
                      <Link className={styles.lessonLink} to={to}>
                        {label}
                      </Link>
                    </li>
                  ))}
                </ul>
              </div>
            ) : null}
          </li>
        );
      })}
    </ul>
  );
}
